# Test-only dependencies for the ML service pytest suite.
# NOT installed in the production image (pytest et al. are absent there by
# design). Install into a GPU-enabled one-off container to run the tests:
#
#   HF=$(grep -E '^HF_TOKEN=' ../../.env.production | cut -d= -f2-)
#   docker run --rm --gpus all --entrypoint /bin/sh \
#     -e HF_TOKEN="$HF" \
#     -v "$PWD":/app -w /app \
#     -v "$PWD/.hf-cache":/home/app/.cache/huggingface \
#     cell-segmentation-hub-ml \
#     -c "pip install -q -r requirements-test.txt && \
#         python -m pytest tests/unit -o addopts='' --asyncio-mode=auto \
#           -m 'not slow' --continue-on-collection-errors \
#           --timeout=60 --timeout-method=thread -p no:cacheprovider -q"
#
# Expected: 228 passed, 1 skipped, 1 deselected, 0 failed, 0 errors.
# GPU is required: models/__init__ imports mamba_ssm/Triton, which fail to
# initialise without a CUDA driver ("0 active drivers").

# pytest 9 requires Python >=3.10, which this image is. Both 8 and 9 dropped
# deprecated APIs, so the floor was moved only after this suite was re-run
# against 9.1.1 on a real GPU -- see the note in
# backend/requirements-pytest-ci.txt, and keep the two floors together.
pytest>=9.1.1
# Floor records reality: >=0.23 already resolved to 1.4.0, so this changes
# nothing at install time — it stops the floor drifting further behind.
pytest-asyncio>=1.4.0
pytest-timeout>=2.2
asgi-lifespan>=2.1
# Was `httpx<0.28`, justified by "starlette 0.27's TestClient passes app= to
# httpx.Client, a kwarg removed in httpx 0.28". That reason is gone: this stack
# pins starlette 1.6.0 / fastapi 0.141.1 (requirements.txt:74-75), a full major
# past the version the comment describes.
#
# Worse, the cap was actively harmful. The ml image ships httpx 0.28.1, so
# `pip install -r requirements-test.txt` DOWNGRADED it at test time — the suite
# ran against an httpx the service never uses. Removing the cap makes the test
# environment match the runtime one.
#
# Measured 2026-09-05, and the cap turned out to be the CAUSE of the 11
# "documented" async-fixture errors this suite carried for months:
#   with httpx<0.28  ->  651 passed, 1 skipped, 11 errors
#   with httpx 0.28.1 ->  662 passed, 1 skipped, 0 errors, 0 failed
# starlette 1.6's TestClient needs the httpx it was written against; pinning
# the test env a major behind the runtime env is what broke the fixtures.
httpx<0.29
